import requests
import json
from typing import Dict, Any, Optional

class APIClient:
    def __init__(self, base_url: str):
        """Initialize API client with base URL."""
        self.base_url = base_url.rstrip('/')
        self.session = requests.Session()
        
    def get_data(self, endpoint: str, params: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """
        Make a GET request to the specified endpoint.
        
        Args:
            endpoint: API endpoint to call
            params: Optional query parameters
            
        Returns:
            JSON response as dictionary
        """
        try:
            # Construct full URL
            url = f"{self.base_url}/{endpoint.lstrip('/')}"
            
            # Make the request
            response = self.session.get(url, params=params)
            
            # Raise an exception for bad status codes
            response.raise_for_status()
            
            # Parse and return JSON response
            return response.json()
            
        except requests.exceptions.RequestException as e:
            print(f"Error making request: {e}")
            raise
        except json.JSONDecodeError as e:
            print(f"Error parsing JSON response: {e}")
            raise

def main():
    # Example using a public API
    api_client = APIClient('https://jsonplaceholder.typicode.com')
    
    try:
        # Get a list of posts
        posts = api_client.get_data('posts')
        print("\nFirst post title:", posts[0]['title'])
        
        # Get a specific post with URL parameters
        post = api_client.get_data('posts', params={'id': 1})
        print("\nPost with ID 1:", json.dumps(post, indent=2))
        
    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    main()
